# Phase 3: GUI — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans. Steps use checkbox (`- [ ]`) syntax.

**Goal:** A one-window PySide6 reader. Open a PDF → page renders immediately → a thumbnail strip → OCR runs on a background thread with a progress bar → a search bar lists hits (page + snippet) → clicking a hit jumps to the page and draws a translucent highlight over the matched text.

**Architecture:** `src/pdf_ocr_reader/gui/` — never imported by the core (Phase 1/2 `test_core_has_no_gui_dep` covers it). `config` reads/writes `%LOCALAPPDATA%\PdfOcrReader\config.toml`. `OcrWorker` (a `QObject` moved to a `QThread`) opens its **own** `Document` for the OCR sweep and emits per-page spans + progress — the main window keeps a separate display-only `Document` for rendering, so two threads never touch one `fitz` handle. `PageGraphicsView` shows the current page as a `QGraphicsPixmapItem` with highlight `QGraphicsRectItem`s; a pure `scene_rect_for_span` function converts a span bbox (PDF points for native text, `ocr.OCR_DPI` pixels for OCR text) into scene coordinates at the current render dpi. `MainWindow` ties it together.

**Tech Stack:** adds `pyside6-essentials>=6.7`. Core unchanged.

**Spec:** `docs/superpowers/specs/2026-09-10-pdf-ocr-reader-design.md`

## Global Constraints

- The core (`document`, `renderer`, `ocr`, `search`, `textlayer`, `cache`, `models`, `cli`) still imports with PySide6 absent. Only `pdf_ocr_reader.gui.*` may import PySide6.
- One `fitz.Document` handle per thread. The main window's `Document` is for rendering + native text only; the worker opens its own for `ocr_all`. They coordinate only through the on-disk cache and Qt signals.
- Exactly one `OcrWorker` runs at a time (RapidOCR's engine is a process-global singleton, not thread-safe). Opening a new file stops the previous worker first.
- Source PDF is opened read-only and never written. "Save searchable copy" is Phase 4 — its menu item is present but disabled in Phase 3.
- Every render call happens on the main thread and is synchronous (fast); OCR never blocks the UI.
- Config: `last_dir` (str), `ocr_dpi` (int, default `ocr.OCR_DPI`), `render_dpi` (int, default 150). Missing file → defaults, no error.

```python
from __future__ import annotations

import os
import tomllib
from dataclasses import asdict, dataclass
from pathlib import Path

from pdf_ocr_reader import ocr

_KEYS = ("last_dir", "ocr_dpi", "render_dpi")


def config_path() -> Path:
    base = os.environ.get("LOCALAPPDATA") or str(Path.home() / "AppData" / "Local")
    return Path(base) / "PdfOcrReader" / "config.toml"


@dataclass
class Config:
    last_dir: str = ""
    ocr_dpi: int = ocr.OCR_DPI
    render_dpi: int = 150


def load_config(path: Path | None = None) -> Config:
    path = path or config_path()
    if not path.exists():
        return Config()
    data = tomllib.loads(path.read_text(encoding="utf-8"))
    return Config(
        last_dir=str(data.get("last_dir", "")),
        ocr_dpi=int(data.get("ocr_dpi", ocr.OCR_DPI)),
        render_dpi=int(data.get("render_dpi", 150)),
    )


def save_config(cfg: Config, path: Path | None = None) -> None:
    path = path or config_path()
    path.parent.mkdir(parents=True, exist_ok=True)
    lines = []
    for key, value in asdict(cfg).items():
        if isinstance(value, str):
            escaped = value.replace("\\", "\\\\").replace('"', '\\"')
            lines.append(f'{key} = "{escaped}"')
        else:
            lines.append(f"{key} = {value}")
    path.write_text("\n".join(lines) + "\n", encoding="utf-8")
```

### Task 2: `OcrWorker`

**Files:**
- Create: `src/pdf_ocr_reader/gui/worker.py`
- Test: `tests/test_gui_worker.py`

**Interfaces:**
- `worker.OcrWorker(QObject)` — constructed with `path: str`. Signals:
  - `pageSpans = Signal(int, object)` — `(page_index, list[Span])`, emitted once per page up front (native or cached), then again for each page as OCR completes.
  - `progress = Signal(int, int)` — `(done, total)` during the OCR sweep.
  - `finished = Signal()`
  - `failed = Signal(str)`
  - method `run()` — opens its own `Document`, emits current spans for every page, runs `ocr_all` (re-emitting each page as it lands), closes, emits `finished`. Any exception → `failed`.
  - method `stop()` — sets a flag checked between pages so a new Open cancels an in-flight sweep.
- `worker.run_in_thread(w: QObject) -> QThread` — moves `w` to a fresh `QThread`, connects `started→run`, `finished`/`failed→quit`, starts it, returns the thread (caller keeps a ref).

```python
from __future__ import annotations

from PySide6.QtCore import QObject, QThread, Signal

from pdf_ocr_reader.document import Document


class OcrWorker(QObject):
    pageSpans = Signal(int, object)
    progress = Signal(int, int)
    finished = Signal()
    failed = Signal(str)

    def __init__(self, path: str) -> None:
        super().__init__()
        self._path = path
        self._stopped = False

    def stop(self) -> None:
        self._stopped = True

    def run(self) -> None:
        try:
            doc = Document.open(self._path)
            for i in range(doc.page_count):
                self.pageSpans.emit(i, doc.page_spans(i, ocr=False))

            pending = doc._pages_needing_ocr()
            total = len(pending)
            for done, page_index in enumerate(pending, start=1):
                if self._stopped:
                    break
                doc._ocr_pages[page_index] = doc._ocr_page(page_index)
                doc._persist_cache()
                self.pageSpans.emit(page_index, doc._ocr_pages[page_index])
                self.progress.emit(done, total)

            doc.close()
            self.finished.emit()
        except Exception as exc:  # noqa: BLE001 - surfaced to the UI
            self.failed.emit(f"{type(exc).__name__}: {exc}")


def run_in_thread(w: QObject) -> QThread:
    thread = QThread()
    w.moveToThread(thread)
    thread.started.connect(w.run)
    w.finished.connect(thread.quit)
    w.failed.connect(thread.quit)
    thread.start()
    return thread
```

(Using `doc._pages_needing_ocr` / `doc._ocr_page` directly lets the worker
check `_stopped` between pages — `Document.ocr_all` has no cancel hook. If a
public cancel is wanted later, add `ocr_all(progress, should_stop)`.)

```python
from __future__ import annotations

from PySide6.QtCore import QByteArray, QRectF, Qt
from PySide6.QtGui import QBrush, QColor, QImage, QPen, QPixmap
from PySide6.QtWidgets import QGraphicsScene, QGraphicsView

_HL_BRUSH = QBrush(QColor(255, 214, 0, 70))
_HL_PEN = QPen(QColor(230, 170, 0, 160))


def scene_rect_for_span(bbox, *, source_dpi: int, render_dpi: int):
    x0, y0, x1, y1 = bbox
    k = render_dpi / source_dpi
    return (x0 * k, y0 * k, (x1 - x0) * k, (y1 - y0) * k)


class PageGraphicsView(QGraphicsView):
    def __init__(self) -> None:
        super().__init__()
        self.setScene(QGraphicsScene(self))
        self.setDragMode(QGraphicsView.ScrollHandDrag)
        self._pixmap_item = None
        self._zoom = 1.0

    def show_page(self, png_bytes: bytes) -> None:
        image = QImage.fromData(QByteArray(png_bytes), "PNG")
        scene = self.scene()
        scene.clear()
        self._pixmap_item = scene.addPixmap(QPixmap.fromImage(image))
        scene.setSceneRect(QRectF(image.rect()))
        self._zoom = 1.0
        self.resetTransform()
        self.fit()

    def set_highlights(self, rects) -> None:
        # remove old highlight rects, keep the pixmap
        for item in list(self.scene().items()):
            if item is not self._pixmap_item:
                self.scene().removeItem(item)
        for x, y, w, h in rects:
            self.scene().addRect(QRectF(x, y, w, h), _HL_PEN, _HL_BRUSH)

    def zoom(self, factor: float) -> None:
        self._zoom *= factor
        self.scale(factor, factor)

    def fit(self) -> None:
        if self._pixmap_item is not None:
            self.fitInView(self._pixmap_item, Qt.KeepAspectRatio)
```

```python
def search_spans(spans_by_page: dict[int, list["Span"]], query: str, *, snippet_chars: int = 160) -> list[PageHit]:
    terms = [t for t in query.lower().split() if t]
    if not terms:
        return []
    hits: list[tuple[int, int, PageHit]] = []
    for page_index, spans in spans_by_page.items():
        text = " ".join(s.text for s in spans)
        lowered = text.lower()
        if not all(term in lowered for term in terms):
            continue
        score = sum(lowered.count(term) for term in terms)
        boxes = [s.bbox for s in spans if any(term in s.text.lower() for term in terms)]
        hits.append((-score, page_index, PageHit(page_index, _snippet(text, terms[0], snippet_chars), boxes)))
    hits.sort(key=lambda t: (t[0], t[1]))
    return [h for _, _, h in hits]
```
